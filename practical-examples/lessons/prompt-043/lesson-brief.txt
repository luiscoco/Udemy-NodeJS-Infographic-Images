import { mkdtemp, writeFile, appendFile, readFile, open, rm } from 'node:fs/promises';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
const dir = await mkdtemp(join(tmpdir(), 'node-lab-'));
const file = join(dir, 'demo.txt');
try {
  await writeFile(file, 'Hello');
  await appendFile(file, ', Node.js!');
  const handle = await open(file, 'r');
  try { console.log(await handle.readFile('utf8')); }
  finally { await handle.close(); }
  await rm(file);
  try { await readFile(file); }
  catch (e) { console.log(e.code); }
} finally { await rm(dir, { recursive: true, force: true }); }
