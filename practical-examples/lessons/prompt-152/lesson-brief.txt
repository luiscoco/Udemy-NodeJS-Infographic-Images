One focused snippet, src/passwords.mjs, exactly:
import { scrypt, randomBytes, timingSafeEqual } from 'node:crypto';
import { promisify } from 'node:util';

const scryptAsync = promisify(scrypt);

export async function hashPassword(password) {
  const salt = randomBytes(16);
  const key = await scryptAsync(password, salt, 64);
  return `${salt.toString('base64')}:${key.toString('base64')}`;
}

export async function verifyPassword(password, stored) {
  const [salt, key] = stored.split(':').map((p) => Buffer.from(p, 'base64'));
  const candidate = await scryptAsync(password, salt, key.length);
  return timingSafeEqual(candidate, key);
}
Teach: a random salt per user; timingSafeEqual for a constant-time comparison; never log passwords, signing keys or bearer tokens (configure the logger to redact the authorization header). A cross-user access attempt returns 404, not 403, so task existence is not revealed. The transactional operation uses one checked-out client with BEGIN, COMMIT and ROLLBACK. Commands, each on one line: npx node-pg-migrate up (reads DATABASE_URL from the environment) and node --test.
