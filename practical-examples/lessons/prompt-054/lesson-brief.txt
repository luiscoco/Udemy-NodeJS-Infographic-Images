import { Transform } from 'node:stream';
import { StringDecoder } from 'node:string_decoder';
class NDJSON extends Transform {
  #decoder = new StringDecoder('utf8'); #pending = '';
  _transform(chunk, encoding, done) {
    try {
      this.#pending += this.#decoder.write(chunk);
      let end;
      while ((end = this.#pending.indexOf('\n')) >= 0) {
        const line = this.#pending.slice(0, end);
        this.#pending = this.#pending.slice(end + 1);
        if (line.length > 1024) throw Error('Record too long');
        if (line.trim()) this.push(JSON.stringify(JSON.parse(line)) + '\n');
      }
      if (this.#pending.length > 1024) throw Error('Record too long');
      done();
    } catch (e) { done(e); }
  }
}
