export function breaker(call, now = Date.now) {
  let failures = 0, until = 0, probing = false;
  return async (...args) => {
    const recovering = until !== 0;
    if (recovering && (now() < until || probing)) throw new Error('OPEN');
    if (recovering) probing = true;
    try {
      const result = await call(...args);
      failures = 0; until = 0;
      return result;
    } catch (error) {
      if (recovering || ++failures >= 3) until = now() + 1000;
      throw error;
    } finally { if (recovering) probing = false; }
  };
}
