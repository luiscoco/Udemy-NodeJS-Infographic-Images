import test from 'node:test';
import assert from 'node:assert/strict';
test('task API persists in PostgreSQL', async t => {
  const app = await buildApp({ pool });
  t.after(() => app.close());
  const title = 'fixture-' + crypto.randomUUID();
  const response = await app.inject({
    method: 'POST', url: '/tasks', payload: { title }
  });
  assert.equal(response.statusCode, 201);
  const created = response.json();
  t.after(() => pool.query('DELETE FROM tasks WHERE id = $1', [created.id]));
  const result = await pool.query('SELECT title FROM tasks WHERE id = $1', [created.id]);
  assert.equal(result.rows[0].title, title);
});
