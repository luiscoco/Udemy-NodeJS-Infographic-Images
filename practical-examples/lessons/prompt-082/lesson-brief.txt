Focused ws server:
import { WebSocketServer } from 'ws';
const server = new WebSocketServer({ port: 8080, maxPayload: 1024 });
server.on('connection', socket => {
  socket.on('error', console.error);
  socket.on('message', data => {
    if (socket.bufferedAmount > 65536) return socket.close(1013, 'Slow client');
    try {
      const message = JSON.parse(data.toString());
      if (typeof message.text !== 'string') return socket.close(1008, 'Invalid message');
      socket.send(JSON.stringify({ text: message.text }));
    } catch { socket.close(1008, 'Invalid JSON'); }
  });
});
