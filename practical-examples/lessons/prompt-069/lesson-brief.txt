import { spawn } from 'node:child_process';
const child = spawn(process.execPath, ['./helper.mjs', 'hello; & echo data'], {
  shell: false, stdio: ['ignore', 'pipe', 'pipe'], timeout: 1000
});
child.stdout.on('data', chunk => process.stdout.write(chunk));
child.stderr.on('data', chunk => process.stderr.write(chunk));
child.on('error', error => console.error(error.code));
child.on('close', (code, signal) => console.log({ code, signal }));
