let stopping = false;
async function shutdown() {
  if (stopping) return;
  stopping = true;
  const deadline = setTimeout(() => {
    server.closeAllConnections();
    process.exit(1);
  }, 10000);
  try {
    await Promise.all([
      new Promise((resolve, reject) =>
        server.close(error => error ? reject(error) : resolve())),
      stopBackground()
    ]);
    await closeDatabase();
    clearTimeout(deadline);
  } catch (error) {
    console.error('Shutdown failed');
    process.exitCode = 1;
  }
}
