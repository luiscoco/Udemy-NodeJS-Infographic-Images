import { setTimeout as delay } from 'node:timers/promises';
export async function retry(operation, signal) {
  const deadline = AbortSignal.any([signal, AbortSignal.timeout(3000)]);
  for (let attempt = 1; attempt <= 4; attempt++) {
    deadline.throwIfAborted();
    try { return await operation({ signal: deadline }); }
    catch (error) {
      deadline.throwIfAborted();
      if (!error.retryable || attempt === 4) throw error;
      const cap = Math.min(1000, 100 * 2 ** (attempt - 1));
      await delay(Math.random() * cap, undefined, { signal: deadline });
    }
  }
}
