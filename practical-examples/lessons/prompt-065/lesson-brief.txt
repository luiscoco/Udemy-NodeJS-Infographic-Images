    const length = pending.readUInt32BE(0);
    if (length > 65536) return socket.destroy(new Error('Frame too large'));
    if (pending.length < 4 + length) break;
    handleMessage(pending.subarray(4, 4 + length));
    pending = pending.subarray(4 + length);
