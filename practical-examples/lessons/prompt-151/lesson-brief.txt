import { createWriteStream } from 'node:fs';
import { rename, rm } from 'node:fs/promises';
import { pipeline } from 'node:stream/promises';

export async function writeAtomically(target, source, signal) {
  const temp = `${target}.${process.pid}.tmp`;
  try {
    await pipeline(source, createWriteStream(temp, { flags: 'wx' }), { signal });
    await rename(temp, target);
  } catch (error) {
    await rm(temp, { force: true });
    throw error;
  }
}
Explain: the output is written to a temporary file next to the target and renamed only on success, so cancellation (Ctrl+C mapped to an AbortController) leaves no partial output; pipeline gives backpressure. Exit codes: 0 success, 1 data error under the fail policy, 2 usage error. Command: node --test.
