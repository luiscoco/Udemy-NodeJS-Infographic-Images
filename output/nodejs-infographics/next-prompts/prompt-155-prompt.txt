const DEADLINE_MS = 10_000;
let stopping = false;

export async function shutdown(signal) {
  if (stopping) return;
  stopping = true;
  app.log.info({ signal }, 'shutting down');
  const timer = setTimeout(() => process.exit(1), DEADLINE_MS);
  timer.unref();
  await app.close();
  await worker.close();
  await pool.end();
  await sdk.shutdown();
  clearTimeout(timer);
}

process.once('SIGTERM', shutdown);
process.once('SIGINT', shutdown);
Teach: readiness returns 503 as soon as stopping is true (see Prompt 139); app.close() stops new connections and waits for in-flight requests; worker.close() lets active jobs finish; sdk.shutdown() flushes telemetry; the deadline is shorter than the Compose stop_grace_period.
