import { Counter, Histogram, Gauge, register } from 'prom-client';
const requests = new Counter({ name: 'app_requests_total', help: 'Requests', labelNames: ['outcome'] });
const duration = new Histogram({ name: 'app_duration_seconds', help: 'Duration' });
const active = new Gauge({ name: 'app_inflight', help: 'Active requests' });
app.get('/work', async () => {
  active.inc();
  const stop = duration.startTimer();
  try {
    const result = await doWork();
    requests.inc({ outcome: 'ok' });
    return result;
  } catch (error) {
    requests.inc({ outcome: 'error' });
    throw error;
  } finally { stop(); active.dec(); }
});
app.get('/metrics', async (_, reply) =>
  reply.type(register.contentType).send(await register.metrics()));
