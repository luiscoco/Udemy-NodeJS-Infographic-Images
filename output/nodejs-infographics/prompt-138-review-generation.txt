services:
  api:
    build: .
    environment:
      PGHOST: db
      PGPORT: 5432
      PGDATABASE: course
      PGUSER: course
      PGPASSWORD: ${DB_PASSWORD:?Set DB_PASSWORD}
    ports: ["3000:3000"]
    depends_on:
      db:
        condition: service_healthy
  db:
    image: postgres:17
    environment:
      POSTGRES_DB: course
      POSTGRES_USER: course
      POSTGRES_PASSWORD: ${DB_PASSWORD:?Set DB_PASSWORD}
    volumes: ["pgdata:/var/lib/postgresql/data"]
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U course -d course"]
      interval: 5s
      timeout: 3s
      retries: 10
volumes:
  pgdata:
