const MAX_ACTIVE = 8;
const MAX_QUEUED = 32;
let active = 0;
const queue = [];
export function submit(work) {
  if (active >= MAX_ACTIVE && queue.length >= MAX_QUEUED) {
    return Promise.reject(new Error('OVERLOADED'));
  }
  return new Promise((resolve, reject) => {
    const run = async () => {
      active++;
      try { resolve(await work()); } catch (error) { reject(error); }
      finally { active--; queue.shift()?.(); }
    };
    if (active < MAX_ACTIVE) void run(); else queue.push(run);
  });
}
